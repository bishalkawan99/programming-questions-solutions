78. Check Palindrome Number

<?php
    $num = 121;
    $original = $num;
    $rev = 0;
    while ($num != 0) {
        $rem = $num % 10;
        $rev = $rev * 10 + $rem;
        $num = intval($num / 10);
    }
    if ($original == $rev)
        echo "$original is Palindrome";
    else
        echo "$original is Not Palindrome";
?>
