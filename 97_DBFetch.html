97. Fetch Data from Database

<?php
    $conn = new mysqli("localhost", "root", "", "mydb");
    
    $sql = "SELECT id, name, email FROM users";
    $result = $conn->query($sql);
    
    if ($result->num_rows > 0) {
        while($row = $result->fetch_assoc()) {
            echo "ID: " . $row["id"] . " Name: " . $row["name"] . 
                 " Email: " . $row["email"] . "<br>";
        }
    }
?>
