31. Count Words in a File

#include <stdio.h>
int main() {
    FILE *fp = fopen("data.txt", "r");
    int wordCount = 0;
    char ch;
    int isWord = 0;
    while ((ch = fgetc(fp)) != EOF) {
        if (ch == ' ' || ch == '\n' || ch == '\t') {
            isWord = 0;
        } else if (!isWord) {
            wordCount++;
            isWord = 1;
        }
    }
    fclose(fp);
    printf("Words = %d\n", wordCount);
    return 0;
}
