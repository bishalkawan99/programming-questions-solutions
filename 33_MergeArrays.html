33. Merge Two Arrays

#include <stdio.h>
int main() {
    int arr1[3] = {1, 2, 3};
    int arr2[3] = {4, 5, 6};
    int merged[6];
    for (int i = 0; i < 3; i++)
        merged[i] = arr1[i];
    for (int i = 0; i < 3; i++)
        merged[3 + i] = arr2[i];
    printf("Merged array: ");
    for (int i = 0; i < 6; i++)
        printf("%d ", merged[i]);
    return 0;
}
