34. Find Second Largest Element

#include <stdio.h>
int main() {
    int arr[5] = {5, 10, 3, 8, 2};
    int max1 = -1, max2 = -1;
    for (int i = 0; i < 5; i++) {
        if (arr[i] > max1) {
            max2 = max1;
            max1 = arr[i];
        } else if (arr[i] > max2 && arr[i] != max1)
            max2 = arr[i];
    }
    printf("Second largest = %d\n", max2);
    return 0;
}
