19. Binary Search

#include <stdio.h>
int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    int search = 30, n = 5;
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = (low + high) / 2;
        if (arr[mid] == search) {
            printf("Element found at index %d\n", mid);
            return 0;
        } else if (arr[mid] < search)
            low = mid + 1;
        else
            high = mid - 1;
    }
    printf("Element not found\n");
    return 0;
}
