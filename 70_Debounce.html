70. Debounce a Function

function debounce(func, delay) {
    let timeoutId;
    return function(...args) {
        clearTimeout(timeoutId);
        timeoutId = setTimeout(() => func(...args), delay);
    };
}
let myFunc = debounce(() => console.log("Called"), 1000);
myFunc();
myFunc();
myFunc();
Output: Called (after 1 second)
