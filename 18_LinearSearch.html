18. Linear Search

#include <stdio.h>
int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    int search = 30, n = 5;
    for (int i = 0; i < n; i++) {
        if (arr[i] == search) {
            printf("Element found at index %d\n", i);
            return 0;
        }
    }
    printf("Element not found\n");
    return 0;
}
